Update `frontend/src/services/api.js` to add:
```js
export async function getForecast(itemId) {
  const response = await fetch(`${API_BASE}/api/forecast?item_id=${encodeURIComponent(itemId)}`);
  if (!response.ok) {
    throw new Error('Failed to fetch forecast data');
  }
  return response.json();
}
```
